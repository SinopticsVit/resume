from __future__ import annotations
import datetime
import logging
import shopify
import typing
import boto3

import os

logger = logging.getLogger()


def callback_handler(event, context):
    params = event["queryStringParameters"]
    shop_url = _get_shop_url(params)
    message = None
    logger.warning(event)
    if shop_url:
        access_token, access_scopes, message = _get_access_credentials(params, shop_url)
        if access_token and access_scopes:
            try:
                _sns_publish(_prepare_payload(shop_url, access_token))
            except Exception as e:
                return _resp(400, f"SNS Publishing Issue Occurred! [400 - {e}]")
            return _resp(200, 'Successfully Published to SNS! [200 - OK]')
    return _resp(400, f'Something Went Wrong! [400 - {message}]')


def _prepare_payload(shop_url: str, access_token: str) -> typing.Dict:
    return {
        "shopUrl": str(shop_url),
        "apiVersion": str(os.environ['SHOPIFY_API_VERSION']),
        "token": str(access_token),
        "shopKey": str(os.environ["SHOPIFY_API_KEY"]),
        "fromDate": datetime.date.today().isoformat(),
        "limit": 10
    }


def _resp(status: int, message: str) -> typing.Dict:
    return {
        "statusCode": status,
        "body": json.dumps(dict(
            message=message
        ))
    }


def _sns_publish(payload: typing.Dict):
    client = boto3.client('sns')
    logger.warning(os.environ['SHOPIFY_AUTH_SNS_TOPIC'])
    client.publish(
        TargetArn=os.environ['SHOPIFY_AUTH_SNS_TOPIC'],
        Message=json.dumps({
            'default': json.dumps(payload)
        }),
        MessageStructure='json'
    )
